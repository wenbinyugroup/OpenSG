% ---------------------------------------------------------------- capability
% trees.  Every capability is drawn with the same vocabulary:
%   inputnode  a file the user edits        (yellow, rounded)
%   funcnode   an OpenSG function that runs (blue, rectangle)
%   datanode   an intermediate quantity     (grey, dashed)
%   outnode    a file written to disk       (green, rounded)
\tikzset{
  capnode/.style={draw, rounded corners=2pt, align=left, font=\footnotesize,
                  inner sep=4pt, minimum height=7mm, text width=#1},
  inputnode/.style={capnode=44mm, fill=yellow!18, draw=yellow!55!black},
  funcnode/.style={capnode=52mm, fill=blue!10, draw=blue!55!black,
                   rounded corners=1pt},
  datanode/.style={capnode=42mm, fill=black!5, draw=black!45, dashed},
  outnode/.style={capnode=46mm, fill=green!12, draw=green!45!black},
  capedge/.style={-{Latex[length=2mm]}, draw=black!62, thick},
  capedge2/.style={capedge, dashed},
}

% A 6x6 or 8x8 matrix can be wider than the text block.  \widematrix shrinks
% a display to the line width ONLY when it would otherwise overrun; narrower
% displays keep their natural size.
\newsavebox{\wmbox}
\newcommand{\widematrix}[1]{%
  \begin{center}%
    \sbox{\wmbox}{$\displaystyle #1$}%
    \ifdim\wd\wmbox>\linewidth
      \resizebox{\linewidth}{!}{\usebox{\wmbox}}%
    \else
      \usebox{\wmbox}%
    \fi
  \end{center}}
